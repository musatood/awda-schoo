رصد نتائج الفصول

1) درجات الطلاب — لحساب متوسط الفصل تلقائياً

اختر الفصل أولاً، ثم أدخل درجة كل طالب (٠-١٠٠) في الجدول، أو استورد ملف Excel/CSV بعمودين: اسم الطالب، الدرجة. لست بحاجة لإضافة الطلاب يدوياً أولاً — أي اسم غير موجود في كشف الفصل سيُضاف تلقائياً من نفس الملف، وتُحسب الدرجات والمتوسط فوراً.

📷 قراءة الصور المكتوبة بخط اليد آلياً (OCR) تقنية غير دقيقة بنسبة ١٠٠٪ خصوصاً مع الخط اليدوي — لذلك بعد رفع الصورة ستظهر لك قائمة الأسماء والدرجات "المقترحة" لمراجعتها وتصحيح أي خطأ يدوياً، ثم تضغط "تأكيد وحفظ" لتُحسب النتيجة. الصور الواضحة والمطبوعة تُعطي نتائج أدق من الخط اليدوي المزدحم.

📄 استيراد Word مخصص لكشف درجات مطبوع بجدول: عمود اسم الطالب، وأعمدة مواد متعددة (مثل تربية إسلامية، لغة عربية، رياضيات...)، وعمود مجموع (اختياري). سيُحوَّل مجموع كل طالب تلقائياً إلى نسبة من 100 بناءً على أعلى درجة ظهرت في كل مادة (سيسألك النظام لتأكيد الحد الأقصى الكلي قبل الحساب)، ثم تظهر لك النتائج في نفس لوحة المراجعة أدناه.

2) حفظ وإرسال تقرير الفصل الشهري

اختر الشهر قبل الحفظ — يمكن رصد نتيجة مستقلة لكل فصل في كل شهر دراسي، وستُحفظ كل شهر بشكل منفصل دون استبدال الأشهر السابقة.

هذا الاستيراد لمتوسط جاهز مسبقاً لكل فصل (بدون درجات فردية) — Excel/CSV: أعمدة (الفصل، المتوسط، ملاحظات). Word: كل سطر بصيغة «اسم الفصل: المتوسط» مثل «أول ابتدائي بنين: 85». سيُسجَّل الاستيراد على الشهر المختار حالياً في حقل "الشهر" أعلاه (القسم 2).

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var out = { status: "success", students: [], finance: {}, admins: [], lessons: [] };
  var sh = ss.getSheetByName("الطلاب");
  if (sh) {
    var rows = sh.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (!rows[i][1]) continue;
      out.students.push({ name: rows[i][1], cls: rows[i][2], phone: String(rows[i][3]) });
    }
  }
  var ls = ss.getSheetByName("دروس_مزامنة");
  if (ls) {
    var lr = ls.getDataRange().getValues();
    for (var k = 1; k < lr.length; k++) {
      if (!lr[k][0]) continue;
      out.lessons.push({ id: String(lr[k][0]), kind: lr[k][1], updatedAt: String(lr[k][2]), json: lr[k][3] });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var sheet;
  if (data.action === "مالية") {
    sheet = ss.getSheetByName("المالية") || ss.insertSheet("المالية");
    sheet.appendRow([new Date(), data.name, data.p1, data.p2, data.p3, data.p4]);
  } else if (data.action === "طلاب") {
    sheet = ss.getSheetByName("الطلاب") || ss.insertSheet("الطلاب");
    sheet.appendRow([new Date(), data.name, data.p1, data.p2]);
  } else if (data.action === "إدارة") {
    sheet = ss.getSheetByName("الإدارة") || ss.insertSheet("الإدارة");
    sheet.appendRow([new Date(), data.name, data.phone]);
  } else if (data.action === "lessons_sync") {
    var lock = LockService.getScriptLock();
    lock.waitLock(20000);
    try {
      sheet = ss.getSheetByName("دروس_مزامنة");
      if (!sheet) { sheet = ss.insertSheet("دروس_مزامنة"); sheet.appendRow(["id", "kind", "updatedAt", "json"]); }
      var f = sheet.createTextFinder(String(data.id)).matchEntireCell(true).findNext();
      if (f && f.getColumn() === 1) {
        sheet.getRange(f.getRow(), 2, 1, 3).setValues([[data.kind, data.updatedAt, data.json]]);
      } else {
        sheet.appendRow([data.id, data.kind, data.updatedAt, data.json]);
      }
    } finally { lock.releaseLock(); }
  }
  return ContentService.createTextOutput("Success");
}
